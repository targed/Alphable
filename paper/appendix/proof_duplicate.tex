\subsection{Formal Proof of Duplicate Tournament Seat and Tile Bias Cancellation}
\label{app:proof_duplicate}

To demonstrate that the paired duplicate tournament format strictly isolates algorithmic superiority from seat advantage and tile draw luck, we formulate the score decomposition theorem.

\begin{theorem}[Exact Algebraic Cancellation of Duplicate Tournament Confounders]
\label{thm:duplicate_cancellation}
Let $\mu_i$ denote the true asymptotic skill level of agent $i \in \{1, 2\}$. Let $\sigma \sim \mathcal{U}(\mathrm{Permutations}(100))$ denote a pseudo-random permutation of the 100-tile Scrabble bag. Then the expected duplicate pair differential satisfies:
\begin{equation}
    \E_\sigma [\Delta_{\mathrm{pair}}(\sigma)] = \mu_1 - \mu_2
\end{equation}
\end{theorem}

\begin{proof}
\textbf{Step 1: Generalized Linear Decomposition of Game Scores.}
In any individual game of tournament Scrabble, the final point score achieved by agent $i$ playing from seat $j \in \{1, 2\}$ under tile bag permutation $\sigma$ can be decomposed into four additive components:
\begin{equation}
    S_{i, j}(\sigma) = \mu_i + \mathrm{Seat}_j + \mathcal{L}_j(\sigma) + \epsilon_{i, j}(\sigma)
\end{equation}
where:
\begin{itemize}
    \item $\mu_i \in \mathbb{R}$ represents the intrinsic algorithmic equity contribution of agent $i$.
    \item $\mathrm{Seat}_j \in \mathbb{R}$ represents the first-mover advantage ($\mathrm{Seat}_1 > \mathrm{Seat}_2$).
    \item $\mathcal{L}_j(\sigma) \in \mathbb{R}$ represents the intrinsic point equity of tiles dealt to seat $j$ under permutation $\sigma$.
    \item $\epsilon_{i, j}(\sigma)$ represents the zero-mean interaction residual ($E_\sigma[\epsilon_{i, j}(\sigma)] = 0$).
\end{itemize}

\textbf{Step 2: Score Margins Across Paired Duplicate Games.}
In a duplicate match on bag sequence $\sigma$:
\begin{enumerate}
    \item \textbf{Game A:} Agent 1 plays from Seat 1; Agent 2 plays from Seat 2. The margin is:
    \begin{align}
        \mathrm{Margin}^{(A)} &= S_{1, 1}^{(A)} - S_{2, 2}^{(A)} \nonumber \\
        &= (\mu_1 - \mu_2) + (\mathrm{Seat}_1 - \mathrm{Seat}_2) + (\mathcal{L}_1(\sigma) - \mathcal{L}_2(\sigma)) + (\epsilon_{1, 1}^{(A)} - \epsilon_{2, 2}^{(A)})
    \end{align}
    \item \textbf{Game B:} Seats are reversed on the identical sequence $\sigma$. Agent 2 plays from Seat 1; Agent 1 plays from Seat 2. The margin (measured from Agent 1's perspective) is:
    \begin{align}
        \mathrm{Margin}^{(B)} &= S_{1, 2}^{(B)} - S_{2, 1}^{(B)} \nonumber \\
        &= (\mu_1 - \mu_2) + (\mathrm{Seat}_2 - \mathrm{Seat}_1) + (\mathcal{L}_2(\sigma) - \mathcal{L}_1(\sigma)) + (\epsilon_{1, 2}^{(B)} - \epsilon_{2, 1}^{(B)})
    \end{align}
\end{enumerate}

\textbf{Step 3: Pair Differential Evaluation.}
The duplicate pair score differential is the arithmetic mean of both games:
\begin{align}
    \Delta_{\mathrm{pair}}(\sigma) &= \frac{\mathrm{Margin}^{(A)} + \mathrm{Margin}^{(B)}}{2} \nonumber \\
    &= (\mu_1 - \mu_2) + \frac{(\mathrm{Seat}_1 - \mathrm{Seat}_2) + (\mathrm{Seat}_2 - \mathrm{Seat}_1)}{2} \nonumber \\
    &\quad + \frac{(\mathcal{L}_1(\sigma) - \mathcal{L}_2(\sigma)) + (\mathcal{L}_2(\sigma) - \mathcal{L}_1(\sigma))}{2} + \bar{\epsilon}(\sigma)
\end{align}

\textbf{Step 4: Algebraic Cancellation.}
Observe the exact cancellations:
\begin{align}
    (\mathrm{Seat}_1 - \mathrm{Seat}_2) + (\mathrm{Seat}_2 - \mathrm{Seat}_1) &\equiv 0 \\
    (\mathcal{L}_1(\sigma) - \mathcal{L}_2(\sigma)) + (\mathcal{L}_2(\sigma) - \mathcal{L}_1(\sigma)) &\equiv 0
\end{align}
Consequently:
\begin{equation}
    \Delta_{\mathrm{pair}}(\sigma) = (\mu_1 - \mu_2) + \bar{\epsilon}(\sigma)
\end{equation}
Taking the expectation over the uniform distribution of bag permutations $\sigma \sim \mathcal{U}(\mathrm{Permutations}(100))$:
\begin{equation}
    \E_\sigma [\Delta_{\mathrm{pair}}(\sigma)] = \mu_1 - \mu_2 + \E_\sigma [\bar{\epsilon}(\sigma)] = \mu_1 - \mu_2
\end{equation}
This concludes the proof: the paired duplicate format algebraically purges both first-mover seat bias and tile allocation luck.
\end{proof}
